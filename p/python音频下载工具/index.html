Python音频下载工具

10月23日, 2026年
阅读时长: 7 分钟

需求:由于部分同事在家办公，又需要访问外网看录屏，国内网络环境难免会遇上视频加载缓慢，请求超时的情况

因此提供两种途径可以暂时满足该需求，原理都是一样向API提交视频下载请求

途径一: 逐一打开录屏链接，F12找到视频链接，在浏览器访问请求下载链接 优点:简单易懂，用户方便操作 缺点:无法看到进度，遇到体积大一点的视频容易下载超时失败

途径二: 由于以前手动分析过网页 Python爬取音频页面分析 这次处理起来相当方便，直接让AI根据文章内容生成图形化界面，再优化一下功能即可制作成工具，并在原来基础上新增了异步/断点/取消/超时自动重试下载功能，用户可以通过F12找到Cookie或Bearer令牌查找视频，再提交下载请求，工具根据定制需求自动重命名 优点:功能扩展多可定制 缺点:用户学习成本增加

"""查询 CXone segment 并下载 MP4；在 Windows 上用 Python 3 运行。"""

import json
import http.client
import os
import queue
import re
import shutil
import socket
import threading
import time as time_module
import urllib.error
import urllib.parse
import urllib.request
import uuid
from datetime import date, datetime, time, timezone
import tkinter as tk
from tkinter import filedialog, messagebox, ttk


API = ""
FIELDS = ["agentNames", "contactNo", "teamNames", "segmentId", "interactionId",
          "acdContactId", "commentText", "hierarchyElementNames"]


class NoRedirect(urllib.request.HTTPRedirectHandler):
    """拒绝重定向，避免将凭据或签名 URL 发送到意外站点。"""

    def redirect_request(self, req, fp, code, msg, headers, newurl):
        raise ValueError("Unexpected HTTP redirect")


OPENER = urllib.request.build_opener(NoRedirect)


def valid_video_url(link):
    parsed = urllib.parse.urlsplit(link)
    host = parsed.hostname or ""
    try:
        return (parsed.scheme == "https" and parsed.port in (None, 443) and
                (host == "s3.amazonaws.com" or host.endswith(".s3.amazonaws.com") or
                 (".s3." in host and host.endswith(".amazonaws.com"))) and
                parsed.path.lower().endswith(".mp4") and not parsed.username and not parsed.password)
    except ValueError:
        return False


def request_json(url, method="GET", body=None, credential="", mode="Cookie"):
    # 凭据仅发送给 CXone API；下载使用播放接口返回的签名 URL。
    headers = {"Accept": "application/json", "Origin": "",
               "Referer": "",
               "originating-service-identifier": "player" if "/media-playback/" in url else "search"}
    if mode == "Cookie":
        headers["Cookie"] = credential
    else:
        headers["Authorization"] = "Bearer " + (credential[7:] if credential.startswith("Bearer ") else credential).strip()
    if body is not None:
        headers["Content-Type"] = "application/json"
    req = urllib.request.Request(url, data=json.dumps(body).encode() if body is not None else None,
                                 headers=headers, method=method)
    with OPENER.open(req, timeout=30) as response:
        return json.load(response)


def search(term, start, end, page, credential, mode):
    # 日期按本机时区确定起止时刻，再转换为接口需要的 UTC 时间。
    start_dt = datetime.combine(date.fromisoformat(start), time.min).astimezone().astimezone(timezone.utc)
    end_dt = datetime.combine(date.fromisoformat(end), time.max).astimezone().astimezone(timezone.utc)
    if start_dt > end_dt:
        raise ValueError("Start date must not be after end date")
    stamp = lambda dt: dt.isoformat(timespec="milliseconds").replace("+00:00", "Z")
    body = {"query": term.strip() or "*", "from": page * 50, "size": 50,
            "filters": [{"field": "startTime", "from": stamp(start_dt), "to": stamp(end_dt)}],
            "facets": [{"field": field, "avoidAggregations": False} for field in
                       ("mediaTypes,recordedMediaFailures", "channelType", "directionType")],
            "sortClauses": [{"field": "startTime", "order": "desc"}],
            "searchOnFields": FIELDS, "isExactMatch": False, "isExport": False,
            "highlights": False}
    result = request_json(API + "/search-service/v1/segments/query", "POST", body, credential, mode)
    return result["results"], result.get("total", 0)


def duration_text(value):
    try:
        seconds = int(value)
        if seconds < 0:
            return ""
    except (ValueError, TypeError):
        return ""
    return "%02d:%02d:%02d" % (seconds // 3600, seconds // 60 % 60, seconds % 60)


def download_filename(segment_id, start_time, agent_names):
    # 搜索结果缺少时间/坐席时使用占位符，避免误写成当前时间。
    try:
        stamp = datetime.strptime(start_time[:19], "%Y-%m-%dT%H:%M:%S").strftime("%Y%m%d_%H%M%S")
    except (TypeError, ValueError):
        stamp = "unknown_time"
    agent = (agent_names or ["unknown"])[0]
    agent = re.sub(r'[<>:"/\\|?*\x00-\x1f]', "_", str(agent)).strip(" .")[:80] or "unknown"
    return "%s_%s_%s.mp4" % (stamp, agent, segment_id)


def video_url(segment_id, credential, mode):
    # 每次重新请求播放信息，取得当前有效的临时下载链接。
    segment_id = str(uuid.UUID(segment_id))
    url = API + "/media-playback/v2/segments/" + segment_id + "?" + urllib.parse.urlencode({
        "media-type": "all", "exclude-waveforms": "false", "exclude-qm-categories": "false"})
    playback = request_json(url, credential=credential, mode=mode)
    for interaction in playback.get("interactions", []):
        link = interaction.get("data", {}).get("fileToPlayUrl")
        if link:
            if not valid_video_url(link):
                raise ValueError("Playback returned an unexpected MP4 URL")
            return link
    raise ValueError("This segment has no downloadable MP4")


class DownloadPaused(Exception):
    pass


def download(segment_id, target, credential, mode, progress=None, pause=None):
    """网络中断时有限次重试；每次获取新链接并从已保存字节续传。"""

    segment_id = str(uuid.UUID(segment_id))
    for attempt in range(4):
        if pause and pause.is_set():
            raise DownloadPaused
        try:
            link = video_url(segment_id, credential, mode)
            _download_once(segment_id, target, link, progress, pause)
            return
        except DownloadPaused:
            raise
        except urllib.error.HTTPError as exc:
            if exc.code not in (408, 429, 500, 502, 503, 504) or attempt == 3:
                raise
            error = exc
        except (TimeoutError, socket.timeout, ConnectionError, urllib.error.URLError,
                http.client.IncompleteRead) as exc:
            if attempt == 3:
                raise OSError("Network timed out after retries; partial file retained for resume") from exc
            error = exc
        # 退避重试可减少连续请求；暂停时立即退出等待。
        delay = 2 ** attempt
        if pause:
            if pause.wait(delay):
                raise DownloadPaused
        else:
            time_module.sleep(delay)


def _download_once(segment_id, target, link, progress=None, pause=None):
    """校验 Range、ETag 和文件长度，成功后才生成最终 MP4。"""

    segment_id = str(uuid.UUID(segment_id))
    part_path = target + ".part"
    meta_path = part_path + ".json"
    try:
        with open(meta_path, encoding="utf-8") as meta_file:
            metadata = json.load(meta_file)
        resumable = metadata.get("segment_id") == segment_id and metadata.get("etag")
        total = int(metadata["total"]) if resumable else 0
        etag = metadata.get("etag") if resumable else None
    except (OSError, ValueError, KeyError, TypeError):
        resumable, total, etag = False, 0, None
    # .part.json 绑定 segment 和文件版本；不要拼接其他录音的残留数据。
    offset = os.path.getsize(part_path) if resumable and os.path.exists(part_path) else 0

    for attempt in range(2):
        if pause and pause.is_set():
            raise DownloadPaused
        headers = {"Range": "bytes=%d-" % offset}
        if offset and etag:
            headers["If-Range"] = etag
        req = urllib.request.Request(link, headers=headers)
        try:
            response_context = OPENER.open(req, timeout=180)
        except urllib.error.HTTPError as exc:
            if offset and exc.code == 416 and attempt == 0:
                offset, total, etag, resumable = 0, 0, None, False
                continue
            raise
        with response_context as response:
            if not valid_video_url(response.url):
                raise ValueError("MP4 was redirected to an unexpected URL")
            if response.status not in (200, 206) or "video/mp4" not in response.headers.get("Content-Type", ""):
                raise ValueError("Server did not return an MP4")
            response_etag = response.headers.get("ETag")
            content_range = response.headers.get("Content-Range", "")
            if response.status == 206:
                match = re.fullmatch(r"bytes (\d+)-(\d+)/(\d+)", content_range)
                if not match:
                    raise ValueError("Invalid video Content-Range")
                start, end, response_total = map(int, match.groups())
                if start != offset or end >= response_total:
                    if offset and attempt == 0:
                        offset, total, etag, resumable = 0, 0, None, False
                        continue
                    raise ValueError("Server returned an unexpected video range")
                chunk_length = end - start + 1
                response_length = response.headers.get("Content-Length")
                if response_length and int(response_length) != chunk_length:
                    raise ValueError("Server returned an inconsistent video range")
                if offset and (response_total != total or (etag and response_etag and etag != response_etag)):
                    if attempt == 0:
                        offset, total, etag, resumable = 0, 0, None, False
                        continue
                    raise ValueError("Video changed while resuming")
                total = response_total
            else:
                offset = 0
                total = int(response.headers.get("Content-Length", "0"))
            if not total or not response_etag:
                raise ValueError("Server omitted video size or ETag; safe resume is unavailable")

            metadata = {"segment_id": segment_id, "etag": response_etag, "total": total}
            meta_temp = meta_path + ".tmp"
            with open(meta_temp, "w", encoding="utf-8") as meta_file:
                json.dump(metadata, meta_file)
            os.replace(meta_temp, meta_path)
            mode = "ab" if offset else "wb"
            received = offset
            if progress:
                progress(received, total)
            with open(part_path, mode) as output:
                while True:
                    if received < total and pause and pause.is_set():
                        raise DownloadPaused
                    chunk = response.read(64 * 1024)
                    if not chunk:
                        break
                    output.write(chunk)
                    received += len(chunk)
                    if progress:
                        progress(received, total)
            if received != total:
                raise ConnectionError("Connection interrupted; resuming partial download")
            break
    else:
        raise ValueError("Could not restart the changed video")

    if os.path.exists(target):
        raise FileExistsError("Target file already exists; partial file retained")
    # 排他创建目标文件，不覆盖已有视频；失败时保留 .part 供下次续传。
    fd = os.open(target, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o666)
    try:
        with os.fdopen(fd, "wb") as output, open(part_path, "rb") as source:
            shutil.copyfileobj(source, output)
    except Exception:
        if os.path.exists(target):
            os.unlink(target)
        raise
    os.unlink(part_path)
    os.unlink(meta_path)

        self.events = queue.Queue()
        self.page = 0

        self.download_active = False
        self.download_paused = False

        ttk.Entry(form, textvariable=self.start, width=12).grid(row=1, column=4)
        ttk.Label(form, text="To").grid(row=1, column=5)

        form.columnconfigure(2, weight=1)
        buttons = ttk.Frame(root, padding=(10, 0))
        buttons.pack(fill="x")
        self.search_button = ttk.Button(buttons, text="Search", command=lambda: self.search_page(0))
        self.search_button.pack(side="left")
        self.next_button = ttk.Button(buttons, text="Next 50", command=lambda: self.search_page(self.page + 1))
        self.next_button.pack(side="left", padx=8)
        self.next_button.state(["disabled"])
        self.search_pause_button = ttk.Button(buttons, text="Pause Search", command=self.toggle_search)
        self.search_pause_button.pack(side="left")
        self.search_pause_button.state(["disabled"])

        self.table = ttk.Treeview(root, columns=("time", "segment", "duration", "agent", "team", "contact", "media"), show="headings")
        for column, title, width in (("time", "Time", 155), ("segment", "Segment ID", 280),
                                     ("duration", "Duration (HH:MM:SS)", 145),
                                     ("agent", "Agent Name", 170), ("team", "Team", 180),
                                     ("contact", "Contact ID", 125), ("media", "Media", 135)):
            self.table.heading(column, text=title)
            self.table.column(column, width=width)
        self.table.pack(fill="both", expand=True, padx=10, pady=8)
        self.table.bind("<<TreeviewSelect>>", self.select_segment)

        bottom = ttk.Frame(root, padding=10)
        bottom.pack(fill="x")
        ttk.Label(bottom, text="Segment ID").pack(side="left")
        ttk.Entry(bottom, textvariable=self.segment, width=42).pack(side="left", padx=5)
        self.download_button = ttk.Button(bottom, text="Download MP4...", command=self.save)
        self.download_button.pack(side="left")
        self.download_pause_button = ttk.Button(bottom, text="Pause Download", command=self.toggle_download)
        self.download_pause_button.pack(side="left", padx=6)
        self.download_pause_button.state(["disabled"])

    def select_segment(self, _event):
        selected = self.table.selection()
        if selected:
            self.segment.set(self.table.item(selected[0], "values")[1])

    def run(self, kind, func):
        # 网络操作在后台线程执行；线程只向队列发送消息，不直接操作 Tk 控件。
        def worker():
            try:
                self.events.put((kind, "ok", func()))
            except DownloadPaused:
                self.events.put((kind, "paused", None))
            except urllib.error.HTTPError as exc:
                self.events.put((kind, "error", f"HTTP {exc.code}; check login, permissions, or expired link"))
            except (OSError, ValueError, KeyError, json.JSONDecodeError) as exc:
                self.events.put((kind, "error", str(exc) if isinstance(exc, (ValueError, OSError)) else type(exc).__name__))

        threading.Thread(target=worker, daemon=True).start()

    def start_search(self):
        self.search_active = True
        self.search_button.state(["disabled"])
        self.next_button.state(["disabled"])
        self.search_pause_button.state(["!disabled"])
        self.run("search", lambda: search(*self.search_args))

    def search_page(self, page):
        credential = self.credential.get().strip()
        if not credential:
            messagebox.showerror("Authentication", "Enter a Cookie or Bearer token first")
            return
        try:
            date.fromisoformat(self.start.get())
            date.fromisoformat(self.end.get())
        except ValueError:
            messagebox.showerror("Date", "Dates must be YYYY-MM-DD")
            return
        self.status.set("Searching...")
        term, start, end, mode = self.term.get(), self.start.get(), self.end.get(), self.mode.get()
        self.search_args = (term, start, end, page, credential, mode)
        self.start_search()

    def toggle_search(self):
        # 单次搜索请求无法中途冻结；暂停期间保留响应，继续后再显示。
        self.search_paused = not self.search_paused
        self.search_pause_button.configure(text="Continue Search" if self.search_paused else "Pause Search")
        if self.search_paused:
            self.status.set("Search paused; an in-flight request may still finish.")
        else:
            self.status.set("Searching...")
            if self.search_result is not None:
                outcome, result = self.search_result
                self.search_result = None
                self.events.put(("search", outcome, result))

    def start_download(self):
        self.download_active = True
        self.download_pause.clear()
        self.download_button.state(["disabled"])
        self.download_pause_button.state(["!disabled"])
        segment_id, target, credential, mode = self.download_args
        last_reported = [0]

        def report_progress(done, total):
            if done == total or done - last_reported[0] >= 256 * 1024:
                last_reported[0] = done
                self.events.put(("progress", "update", (done, total)))

        self.run("download", lambda: download(segment_id, target, credential, mode, report_progress,
                                                self.download_pause))

    def toggle_download(self):
        # 下载在下一个数据块边界停止，保留 .part 和校验信息。
        self.download_paused = not self.download_paused
        self.download_pause_button.configure(text="Continue Download" if self.download_paused else "Pause Download")
        if self.download_paused:
            self.download_pause.set()
            self.status.set("Pausing download after the current chunk; partial file is retained.")
        elif self.download_active:
            self.download_pause.clear()
            self.status.set("Downloading...")
        else:
            self.status.set("Downloading...")
            self.start_download()

    def save(self):
        try:
            segment_id = str(uuid.UUID(self.segment.get().strip()))
        except ValueError:
            messagebox.showerror("Segment ID", "Enter a valid segment UUID")
            return
        credential = self.credential.get().strip()
        if not credential:
            messagebox.showerror("Authentication", "Enter a Cookie or Bearer token first")
            return
        row = self.search_rows.get(segment_id, {})
        filename = download_filename(segment_id, row.get("startTime"), row.get("agentNames"))
        target = filedialog.asksaveasfilename(defaultextension=".mp4", initialfile=filename,
                                              filetypes=[("MP4 video", "*.mp4")])
        if not target:
            return
        if os.path.exists(target):
            messagebox.showerror("File exists", "Choose a new filename; existing files are not overwritten")
            return
        self.progress_bar["value"] = 0
        self.progress_text.set("Starting...")
        self.status.set("Downloading; interrupted downloads can be resumed by choosing the same segment and destination.")
        mode = self.mode.get()
        self.download_args = (segment_id, target, credential, mode)
        self.start_download()

    def poll(self):
        # 所有进度和结果均在 Tk 主线程中更新。
        try:
            kind, outcome, result = self.events.get_nowait()
        except queue.Empty:
            self.root.after(100, self.poll)
            return
        if kind == "progress":
            done, total = result
            self.progress_bar["value"] = done * 100 / total
            self.progress_text.set("%d%%  %.1f / %.1f MB" % (done * 100 / total, done / 1048576, total / 1048576))
            self.root.after(100, self.poll)
            return
        if kind == "search":
            self.search_active = False
            if self.search_paused:
                self.search_result = (outcome, result)
                self.status.set("Search paused; continue to show this page.")
            else:
                self.search_button.state(["!disabled"])
                self.search_pause_button.state(["disabled"])
                if outcome == "error":
                    self.status.set(result)
                    messagebox.showerror("Search failed", result)
                else:
                    rows, self.total = result
                    self.page = self.search_args[3]
                    self.search_rows = {row["segmentId"]: row for row in rows if row.get("segmentId")}
                    self.table.delete(*self.table.get_children())
                    for row in rows:
                        self.table.insert("", "end", values=(
                            row.get("startTime", ""), row.get("segmentId", ""), duration_text(row.get("duration")),
                            ", ".join(row.get("agentNames") or []), ", ".join(row.get("teamNames") or []),
                            row.get("acdContactId", ""), ", ".join(row.get("mediaTypes") or [])))
                    self.status.set(f"Page {self.page + 1}: {len(rows)} results (server total: {self.total})")
                self.next_button.state(["!disabled"] if (self.page + 1) * 50 < self.total else ["disabled"])
        else:
            self.download_active = False
            if outcome == "paused":
                if not self.download_paused:
                    self.start_download()
                else:
                    self.status.set("Download paused; continue to resume from the saved part.")
            else:
                self.download_paused = False
                self.download_pause_button.configure(text="Pause Download")
                self.download_pause_button.state(["disabled"])
                self.download_button.state(["!disabled"])
                if outcome == "error":
                    part = self.download_args[1] + ".part"
                    self.status.set(result + (" Partial data retained: " + part if os.path.exists(part) else ""))
                    messagebox.showerror("Download failed", self.status.get())
                else:
                    self.status.set("Saved: " + self.download_args[1])
                    self.progress_bar["value"] = 100
                    self.progress_text.set("100%  Complete")
        self.root.after(100, self.poll)


if __name__ == "__main__":
    App(tk.Tk()).root.mainloop()
