初学Playwright

from playwright.sync_api import sync_playwright
import time

# 查看有哪些frame
def dump_frame_tree(frame, indent):
    print(indent + frame.name + '@' + frame.url)
    for child in frame.child_frames:
        dump_frame_tree(child, indent + "    ")

# 同步模式（新手首选）
# 启动Chrome浏览器，headless=False表示显示浏览器窗口（调试用）
# user_data_dir保留用户登录cookie
# start-maximized窗口最大化，注意元素定位有加载速度问题
p = sync_playwright().start()
browser = p.chromium.launch_persistent_context(
    user_data_dir="./playwright_profile",
    headless=False,
    args=['--start-maximized']
)
page = browser.new_page()
page.goto("",wait_until="networkidle")

frame = page.frame_locator('[title="Document reading pane"]').frame_locator('[title="Inner document reading pane"]')
text = frame.locator("body").inner_text()
print(frame)
print(text)
page.pause()

with open("extracted_content.txt", "a+", encoding="utf-8") as f:   # a+ 追加，多次抓取不覆盖
    for url in urls:
        try:
            page.goto(url,wait_until="networkidle")   # 等网络空闲，确保内容已渲染
            # 正文在两层嵌套 iframe 内，需逐层定位
            frame = page.frame_locator('[title="Document reading pane"]').frame_locator('[title="Inner document reading pane"]')
            text = frame.locator("body").inner_text()   # 取 iframe 内的纯文本
            time.sleep(1)                              # 稍作停顿，避免请求过快
            f.write(text+'\n')
            print(f"\n✓ 已保存 {url} 页面的内容")
        except Exception as e:
            print(f"{url} 页面 失败: {e}")   # 单页失败不中断整轮循环
    f.close()
